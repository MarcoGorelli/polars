1//! Rewrites of `[NOT] EXISTS` / `[NOT] IN (subquery)` predicates into semi /
2//! anti joins, decorrelating equi-correlation predicates into join keys.
3//! Subquery shapes the rewrites can't soundly express return `None` so the
4//! caller falls back to the generic filter path.

26impl SQLContext {
27    // Entry point: offer each WHERE conjunct to the rewrite, returning the
28    // (possibly join-extended) frame together with the conjuncts left for the
29    // ordinary filter path. In `KeepTrue` mode each top-level AND-conjunct is
30    // offered independently. In `RemoveTrue` mode conjuncts can't be split
31    // (`NOT (a AND b)` is a disjunction), so only a sole (possibly
32    // parenthesized) subquery predicate rewrites, and anything else is
33    // returned whole as the residual.
34    pub(crate) fn rewrite_subquery_conjuncts<'a>(
35        &mut self,
36        mut lf: LazyFrame,
37        expr: &'a SQLExpr,
38        filter_mode: FilterMode,
39        schema: &Schema,
40    ) -> PolarsResult<(LazyFrame, Vec<&'a SQLExpr>)> {
41        let residual = match filter_mode {
42            FilterMode::RemoveTrue => {
43                let mut unwrapped = expr;
44                while let SQLExpr::Nested(inner) = unwrapped {
45                    unwrapped = inner;
46                }
47                match self.try_rewrite_subquery_conjunct(&lf, unwrapped, filter_mode, schema)? {
48                    Some(new_lf) => {
49                        lf = new_lf;
50                        Vec::new()
51                    },
52                    None => vec![expr],
53                }
54            },
55            FilterMode::KeepTrue => {
56                let mut residual = Vec::new();
57                for conj in MintermIter::new(expr) {
58                    if let Some(new_lf) =
59                        self.try_rewrite_subquery_conjunct(&lf, conj, filter_mode, schema)?
60                    {
61                        lf = new_lf;
62                    } else {
63                        residual.push(conj);
64                    }
65                }
66                residual
67            },
68        };
69        Ok((lf, residual))
70    }

72    // Dispatch one conjunct to the matching rewrite. `RemoveTrue` mode (DELETE)
73    // flips the join polarity. Removing `IN` rows keeps
74    // rows whose membership is false or NULL, exactly what an anti-join
75    // produces; removing `NOT IN` rows would additionally keep NULL keys, which
76    // a semi join can't express, so it stays on the filter path.
77    fn try_rewrite_subquery_conjunct(
78        &mut self,
79        lf: &LazyFrame,
80        conj: &SQLExpr,
81        filter_mode: FilterMode,
82        schema: &Schema,
83    ) -> PolarsResult<Option<LazyFrame>> {
84        let removing = filter_mode == FilterMode::RemoveTrue;
85        match conj {
86            SQLExpr::Exists { subquery, negated } => {
87                self.try_rewrite_exists_as_join(lf, subquery, *negated != removing, schema)
88            },
89            SQLExpr::InSubquery {
90                expr: lhs,
91                subquery,
92                negated,
93            } if !(*negated && removing) => self.try_rewrite_in_subquery_as_join(
94                lf,
95                lhs,
96                subquery,
97                *negated != removing,
98                filter_mode,
99                schema,
100            ),
101            _ => Ok(None),
102        }
103    }

105    // Lower `[NOT] EXISTS (SELECT ... FROM rel WHERE rel.k = outer.k ...)` to a
106    // semi / anti join by decorrelating the equi-correlation predicate(s) into
107    // join keys. DISTINCT is ignored: existence is invariant under
108    // deduplication. A non-equality correlation predicate falls back to the
109    // count-filter rewrite below.
110    #[cfg(feature = "semi_anti_join")]
111    fn try_rewrite_exists_as_join(
112        &mut self,
113        lf: &LazyFrame,
114        subquery: &Query,
115        negated: bool,
116        outer_schema: &Schema,
117    ) -> PolarsResult<Option<LazyFrame>> {
118        let Some(select) = eligible_subquery_select(subquery) else {
119            return Ok(None);
120        };
121        let Some(selection) = &select.selection else {
122            return Ok(None);
123        };
124        // Resolve and parse the inner relation in an isolated context so its
125        // table/alias registrations don't leak into the outer query's scope.
126        let mut ctx = self.isolated();
127        let Some((inner_names, inner_lf, inner_schema)) =
128            ctx.resolve_subquery_from(&select.from[0])?
129        else {
130            return Ok(None);
131        };
132        if let Some(SubqueryConjuncts {
133            left_on,
134            right_on,
135            local_filters,
136        }) =
137            ctx.split_subquery_conjuncts(selection, &inner_names, &inner_schema, outer_schema)?
138        {
139            // An uncorrelated EXISTS (no correlation key found) has no join key
140            // to build from, so leave it to the existing path.
141            return Ok(if left_on.is_empty() {
142                None
143            } else {
144                Some(ctx.finish_decorrelated_join(
145                    lf,
146                    inner_lf,
147                    left_on,
148                    right_on,
149                    local_filters,
150                    negated,
151                )?)
152            });
153        }
154        ctx.try_rewrite_exists_as_count_filter(
155            lf,
156            inner_lf,
157            selection,
158            &inner_names,
159            &inner_schema,
160            outer_schema,
161            negated,
162        )
163    }

165    // Lower `[NOT] EXISTS (SELECT ... FROM rel WHERE <comparison-correlated>)`,
166    // whose correlation predicates aren't all plain equalities, to a
167    // `count(*) > 0` filter (`== 0` for NOT EXISTS).
168    #[cfg(feature = "semi_anti_join")]
169    #[expect(clippy::too_many_arguments)]
170    fn try_rewrite_exists_as_count_filter(
171        mut self,
172        lf: &LazyFrame,
173        inner_lf: LazyFrame,
174        selection: &SQLExpr,
175        inner_names: &PlHashSet<String>,
176        inner_schema: &Schema,
177        outer_schema: &Schema,
178        negated: bool,
179    ) -> PolarsResult<Option<LazyFrame>> {
180        let mut corr_preds = Vec::new();
181        let mut local_filters = Vec::new();
182        for conj in MintermIter::new(selection) {
183            if let Some(pred) =
184                scalar_correlation_predicate(conj, inner_names, inner_schema, outer_schema)
185            {
186                corr_preds.push(pred);
187            } else if let Some(filter) =
188                self.try_parse_inner_only_expr(conj, inner_names, inner_schema)?
189            {
190                local_filters.push(filter);
191            } else {
192                return Ok(None);
193            }
194        }
195        // No correlation: leave it to the equi-only path.
196        if corr_preds.is_empty() {
197            return Ok(None);
198        }
199
200        let prefix = format_pl_smallstr!("{CORRELATED_COL_PREFIX}{}_", unique_column_name());
201        let idx_name = format_pl_smallstr!("{prefix}idx");
202        let count_name = format_pl_smallstr!("{prefix}cnt");
203
204        // Apply inner-only filters, then rename inner columns to collision-free
205        // names so they can't clash with outer columns of the same name (as in
206        // a self-correlated `t1 AS x` subquery over outer `t1`).
207        let inner_filtered = local_filters.into_iter().fold(inner_lf, LazyFrame::filter);
208        let rename_from: Vec<PlSmallStr> = inner_schema.iter_names().cloned().collect();
209        let rename_to: Vec<PlSmallStr> = rename_from
210            .iter()
211            .map(|name| prefixed_inner(&prefix, name))
212            .collect();
213        let inner_renamed = inner_filtered.rename(&rename_from, &rename_to, true);
214        inner_renamed.set_cached_arena(self.lp_arena, self.expr_arena);
215
216        let join_preds: Vec<Expr> = corr_preds.iter().map(|p| p.to_expr(&prefix)).collect();
217
218        let outer_indexed = row_indexed_once(lf.clone(), idx_name.clone());
219        let matched = outer_indexed
220            .clone()
221            .join_builder()
222            .with(inner_renamed)
223            .how(JoinType::Inner)
224            .join_where(join_preds);
225        let grouped = matched
226            .group_by([col(idx_name.clone())])
227            .agg([len().alias(count_name.clone())]);
228
229        let joined = outer_indexed
230            .join_builder()
231            .with(grouped)
232            .left_on([col(idx_name.clone())])
233            .right_on([col(idx_name.clone())])
234            .how(JoinType::Left)
235            .coalesce(JoinCoalesce::CoalesceColumns)
236            .maintain_order(MaintainOrderJoin::Left)
237            .finish()?
238            .with_columns([col(count_name.clone()).fill_null(lit(0))]);
239
240        let matches = if negated {
241            col(count_name.clone()).eq(lit(0))
242        } else {
243            col(count_name.clone()).gt(lit(0))
244        };
245        Ok(Some(joined.filter(matches).drop(Selector::ByName {
246            names: Arc::from([idx_name, count_name]),
247            strict: true,
248        })))
249    }

251    // Lower `lhs [NOT] IN (SELECT col FROM rel ...)` to a semi / anti join: the
252    // projected column is the membership key and any equi-correlations in the
253    // subquery WHERE become additional join keys.
254    #[cfg(feature = "semi_anti_join")]
255    fn try_rewrite_in_subquery_as_join(
256        &mut self,
257        lf: &LazyFrame,
258        lhs: &SQLExpr,
259        subquery: &Query,
260        anti: bool,
261        filter_mode: FilterMode,
262        outer_schema: &Schema,
263    ) -> PolarsResult<Option<LazyFrame>> {
264        let Some(select) = eligible_subquery_select(subquery) else {
265            return Ok(None);
266        };
267        // DISTINCT is membership-invariant, but DISTINCT ON drops rows per key.
268        if matches!(&select.distinct, Some(Distinct::On(_))) {
269            return Ok(None);
270        }
271        let [SelectItem::UnnamedExpr(proj) | SelectItem::ExprWithAlias { expr: proj, .. }] =
272            select.projection.as_slice()
273        else {
274            return Ok(None);
275        };
276
277        let Some(left_key) = self.try_parse_outer_only_expr(lhs, outer_schema)? else {
278            return Ok(None);
279        };
280
281        let mut ctx = self.isolated();
282        let Some((inner_names, inner_lf, inner_schema)) =
283            ctx.resolve_subquery_from(&select.from[0])?
284        else {
285            return Ok(None);
286        };
287        // The membership key must be a plain expression over the inner relation;
288        // any alias it carries is cosmetic and not allowed in a join key.
289        let Some(right_key) = ctx.try_parse_inner_only_expr(proj, &inner_names, &inner_schema)?
290        else {
291            return Ok(None);
292        };
293        let right_key = right_key.meta().undo_aliases();
294        if !usable_as_join_key(&left_key) || !usable_as_join_key(&right_key) {
295            return Ok(None);
296        }
297
298        let SubqueryConjuncts {
299            mut left_on,
300            mut right_on,
301            local_filters,
302        } = match &select.selection {
303            Some(selection) => {
304                let Some(split) = ctx.split_subquery_conjuncts(
305                    selection,
306                    &inner_names,
307                    &inner_schema,
308                    outer_schema,
309                )?
310                else {
311                    return Ok(None);
312                };
313                split
314            },
315            None => SubqueryConjuncts::default(),
316        };
317
318        // Correlation keys for the "NOT IN" 3VL correction
319        let corr_outer = left_on.clone();
320        let corr_inner = right_on.clone();
321        left_on.insert(0, left_key.clone());
322        right_on.insert(0, right_key.clone());
323
324        // Inline, so filtered inner frame can be reused for the correction
325        let inner_lf = local_filters.into_iter().fold(inner_lf, LazyFrame::filter);
326        inner_lf.set_cached_arena(ctx.lp_arena, ctx.expr_arena);
327        let joined = build_semi_anti_join(lf, inner_lf.clone(), left_on, right_on, anti)?;
328
329        // Only `KeepTrue` "NOT IN" needs 3VL correction.
330        if !(anti && filter_mode == FilterMode::KeepTrue) {
331            return Ok(Some(joined));
332        }
333        Ok(Some(refine_not_in_anti_join(
334            joined,
335            inner_lf,
336            &left_key,
337            &right_key,
338            &corr_outer,
339            &corr_inner,
340        )?))
341    }

343    // Apply the local filters to the inner relation, hand this (isolated, now
344    // finished) context's arenas to it, and build the semi / anti join against
345    // the outer frame. Consumes the context: nothing may be parsed in the
346    // subquery's scope after the join is built.
347    #[cfg(feature = "semi_anti_join")]
348    fn finish_decorrelated_join(
349        self,
350        lf: &LazyFrame,
351        inner_lf: LazyFrame,
352        left_on: Vec<Expr>,
353        right_on: Vec<Expr>,
354        local_filters: Vec<Expr>,
355        anti: bool,
356    ) -> PolarsResult<LazyFrame> {
357        let inner_lf = local_filters.into_iter().fold(inner_lf, LazyFrame::filter);
358        inner_lf.set_cached_arena(self.lp_arena, self.expr_arena);
359        build_semi_anti_join(lf, inner_lf, left_on, right_on, anti)
360    }

362    // Resolve the subquery's FROM (a single relation, possibly with joins) into
363    // the inner LazyFrame, its schema, and the set of relation names/aliases
364    // used to classify qualified correlation columns.
365    fn resolve_subquery_from(
366        &mut self,
367        tbl_expr: &TableWithJoins,
368    ) -> PolarsResult<Option<(PlHashSet<String>, LazyFrame, SchemaRef)>> {
369        let Some(inner_names) = std::iter::once(&tbl_expr.relation)
370            .chain(tbl_expr.joins.iter().map(|j| &j.relation))
371            .map(get_table_name)
372            .collect::<Option<PlHashSet<_>>>()
373        else {
374            return Ok(None);
375        };
376        let mut inner_lf = self.execute_from_statement(tbl_expr)?;
377        let inner_schema = self.get_frame_schema(&mut inner_lf)?;
378        Ok(Some((inner_names, inner_lf, inner_schema)))
379    }

381    // Split a subquery's WHERE conjuncts into a `SubqueryConjuncts`, or `None`
382    // when a conjunct is neither a correlation key pair nor an inner-only
383    // filter (an outer column in a non-equi shape, an unresolvable name).
384    #[cfg(feature = "semi_anti_join")]
385    fn split_subquery_conjuncts(
386        &mut self,
387        selection: &SQLExpr,
388        inner_names: &PlHashSet<String>,
389        inner_schema: &Schema,
390        outer_schema: &Schema,
391    ) -> PolarsResult<Option<SubqueryConjuncts>> {
392        let mut left_on = Vec::new();
393        let mut right_on = Vec::new();
394        let mut local_filters = Vec::new();
395        for conj in MintermIter::new(selection) {
396            if let Some((outer_key, inner_key)) =
397                correlation_key_pair(conj, inner_names, inner_schema, outer_schema)
398            {
399                left_on.push(col(outer_key));
400                right_on.push(col(inner_key));
401                continue;
402            }
403            let Some(filter) = self.try_parse_inner_only_expr(conj, inner_names, inner_schema)?
404            else {
405                return Ok(None);
406            };
407            local_filters.push(filter);
408        }
409        Ok(Some(SubqueryConjuncts {
410            left_on,
411            right_on,
412            local_filters,
413        }))
414    }

416    // Parse an outer-query expression, or `None` if it doesn't stand on the outer
417    // relation alone: an unlowered subquery, or a column the outer frame lacks.
418    // Any alias is cosmetic here and stripped.
419    fn try_parse_outer_only_expr(
420        &mut self,
421        sql_expr: &SQLExpr,
422        outer_schema: &Schema,
423    ) -> PolarsResult<Option<Expr>> {
424        let expr = parse_sql_expr(sql_expr, self, Some(outer_schema))?
425            .meta()
426            .undo_aliases();
427        if has_expr(&expr, |e| matches!(e, Expr::SubPlan(_, _)))
428            || !expr_to_leaf_column_names_iter(&expr)
429                .all(|name| outer_schema.contains(name.as_str()))
430        {
431            return Ok(None);
432        }
433        Ok(Some(expr))
434    }

436    // Parse a subquery expression as one over the inner relation only, or `None`
437    // if it references any outer column (a correlation shape we don't handle) or
438    // contains a nested subquery.
439    fn try_parse_inner_only_expr(
440        &mut self,
441        sql_expr: &SQLExpr,
442        inner_names: &PlHashSet<String>,
443        inner_schema: &Schema,
444    ) -> PolarsResult<Option<Expr>> {
445        if !binds_to_inner_relation(sql_expr, inner_names, inner_schema) {
446            return Ok(None);
447        }
448        Ok(Some(parse_sql_expr(sql_expr, self, Some(inner_schema))?))
449    }

451    // Lower every correlated subquery reachable from `expr` into a decorrelated
452    // join over `lf`, substituting each one for the column its lowering
453    // materialised. Returns the updated frame together with the rewritten
454    // expression, which borrows `expr` unchanged when nothing was lowered.
455    // Subqueries that can't be lowered are left in place.
456    //
457    // See [`SubqueryBindings`] for when a set may be shared across calls.
458    pub(crate) fn lower_correlated_subqueries<'a>(
459        &mut self,
460        lf: LazyFrame,
461        outer_schema: &Schema,
462        expr: &'a SQLExpr,
463        scope: LowerScope,
464        bindings: &mut SubqueryBindings,
465    ) -> PolarsResult<(LazyFrame, Cow<'a, SQLExpr>)> {
466        // The mutable visit needs an owned expression; skip the clone when there is
467        // nothing to rewrite.
468        if !expr_contains_subquery(expr) {
469            return Ok((lf, Cow::Borrowed(expr)));
470        }
471        let mut rewritten = expr.clone();
472        let mut lowering = CorrelatedLowering {
473            ctx: self,
474            lf,
475            outer_schema,
476            scope,
477            bindings,
478            query_depth: 0,
479            changed: false,
480        };
481        if let ControlFlow::Break(e) = VisitMut::visit(&mut rewritten, &mut lowering) {
482            return Err(e);
483        }
484        let rewritten = if lowering.changed {
485            Cow::Owned(rewritten)
486        } else {
487            Cow::Borrowed(expr)
488        };
489        Ok((lowering.lf, rewritten))
490    }
491
492    // Attempt the decorrelation of a single scalar aggregate subquery:
493    //   (SELECT AGG(...) FROM inner WHERE <corr-preds> AND <inner-filters>)
494    // Equality-only correlation: `inner GROUP BY <corr cols>` once, joined onto
495    // the outer frame on those columns directly. Otherwise: row-index the outer
496    // frame, inner-join outer × inner on the correlation predicates (`join_where`
497    // handles inequality), aggregate per outer row, then left-join back on the
498    // row index. Either way, `COUNT` over no matches is 0, every other aggregate
499    // is NULL. Returns `None` when uncorrelated or not a scalar-aggregate shape.
500    fn try_decorrelate_scalar_subquery(
501        &mut self,
502        lf: LazyFrame,
503        outer_schema: &Schema,
504        subquery: &Query,
505    ) -> PolarsResult<Option<(LazyFrame, PlSmallStr)>> {
506        let Some(select) = eligible_subquery_select(subquery) else {
507            return Ok(None);
508        };
509        let Some(selection) = &select.selection else {
510            return Ok(None);
511        };
512        let [SelectItem::UnnamedExpr(proj) | SelectItem::ExprWithAlias { expr: proj, .. }] =
513            select.projection.as_slice()
514        else {
515            return Ok(None);
516        };
517
518        let mut ctx = self.isolated();
519        let Some((inner_names, inner_lf, inner_schema)) =
520            ctx.resolve_subquery_from(&select.from[0])?
521        else {
522            return Ok(None);
523        };
524
525        // The projection must be a scalar aggregate over the inner relation.
526        let Some(agg_expr) = ctx.try_parse_inner_only_expr(proj, &inner_names, &inner_schema)?
527        else {
528            return Ok(None);
529        };
530        if !has_expr(&agg_expr, |e| matches!(e, Expr::Agg(_) | Expr::Len)) {
531            return Ok(None);
532        }
533        let count_like = matches!(
534            agg_output_root(&agg_expr),
535            Expr::Len | Expr::Agg(AggExpr::Count { .. })
536        );
537
538        // Split the WHERE into correlation predicates and inner-only filters.
539        let mut corr_preds = Vec::new();
540        let mut local_filters = Vec::new();
541        for conj in MintermIter::new(selection) {
542            if let Some(pred) =
543                scalar_correlation_predicate(conj, &inner_names, &inner_schema, outer_schema)
544            {
545                corr_preds.push(pred);
546            } else if let Some(filter) =
547                ctx.try_parse_inner_only_expr(conj, &inner_names, &inner_schema)?
548            {
549                local_filters.push(filter);
550            } else {
551                return Ok(None);
552            }
553        }
554        // No correlation: leave it to the uncorrelated scalar-subquery path.
555        if corr_preds.is_empty() {
556            return Ok(None);
557        }
558
559        let prefix = format_pl_smallstr!("{CORRELATED_COL_PREFIX}{}_", unique_column_name());
560        let result_name = format_pl_smallstr!("{prefix}res");
561
562        // Apply inner-only filters, then rename inner columns to collision-free
563        // names so they can't clash with outer columns of the same name (as in a
564        // self-correlated `t1 AS x` subquery over outer `t1`).
565        let inner_filtered = local_filters.into_iter().fold(inner_lf, LazyFrame::filter);
566        let rename_from: Vec<PlSmallStr> = inner_schema.iter_names().cloned().collect();
567        let rename_to: Vec<PlSmallStr> = rename_from
568            .iter()
569            .map(|name| prefixed_inner(&prefix, name))
570            .collect();
571        let inner_renamed = inner_filtered.rename(&rename_from, &rename_to, true);
572        inner_renamed.set_cached_arena(ctx.lp_arena, ctx.expr_arena);
573
574        let agg_expr = agg_expr.map_expr(|e| match e {
575            Expr::Column(name) if inner_schema.contains(name.as_str()) => {
576                col(prefixed_inner(&prefix, &name))
577            },
578            other => other,
579        });
580
581        if corr_preds.iter().all(|p| p.op == SQLBinaryOperator::Eq) {
582            let outer_on: Vec<Expr> = corr_preds.iter().map(|p| col(p.outer.clone())).collect();
583            let inner_on: Vec<Expr> = corr_preds
584                .iter()
585                .map(|p| col(prefixed_inner(&prefix, &p.inner)))
586                .collect();
587            let grouped = inner_renamed
588                .group_by(inner_on.clone())
589                .agg([agg_expr.alias(result_name.clone())]);
590            let joined =
591                left_join_aggregate(lf, grouped, outer_on, inner_on, &result_name, count_like)?;
592            return Ok(Some((joined, result_name)));
593        }
594
595        let join_preds: Vec<Expr> = corr_preds.iter().map(|p| p.to_expr(&prefix)).collect();
596        let idx_name = format_pl_smallstr!("{prefix}idx");
597
598        let outer_indexed = row_indexed_once(lf, idx_name.clone());
599        let matched = outer_indexed
600            .clone()
601            .join_builder()
602            .with(inner_renamed)
603            .how(JoinType::Inner)
604            .join_where(join_preds);
605        let grouped = matched
606            .group_by([col(idx_name.clone())])
607            .agg([agg_expr.alias(result_name.clone())]);
608
609        let joined = left_join_aggregate(
610            outer_indexed,
611            grouped,
612            vec![col(idx_name.clone())],
613            vec![col(idx_name.clone())],
614            &result_name,
615            count_like,
616        )?
617        .drop(Selector::ByName {
618            names: Arc::from([idx_name]),
619            strict: true,
620        });
621        Ok(Some((joined, result_name)))
622    }
623
624    // Attempt the decorrelation of a correlated `IN (subquery)` into a boolean
625    // column. The correlated candidate values are collected per outer row into a
626    // list, and membership is then evaluated against that list under SQL's
627    // three-valued logic. An uncorrelated subquery is left to the generic path.
628    fn try_decorrelate_in_subquery(
629        &mut self,
630        lf: LazyFrame,
631        outer_schema: &Schema,
632        lhs: &SQLExpr,
633        subquery: &Query,
634    ) -> PolarsResult<Option<(LazyFrame, PlSmallStr)>> {
635        let Some(select) = eligible_subquery_select(subquery) else {
636            return Ok(None);
637        };
638        if matches!(&select.distinct, Some(Distinct::On(_))) {
639            return Ok(None);
640        }
641        let [SelectItem::UnnamedExpr(proj) | SelectItem::ExprWithAlias { expr: proj, .. }] =
642            select.projection.as_slice()
643        else {
644            return Ok(None);
645        };
646        let Some(selection) = &select.selection else {
647            return Ok(None);
648        };
649
650        let Some(needle) = self.try_parse_outer_only_expr(lhs, outer_schema)? else {
651            return Ok(None);
652        };
653
654        let mut ctx = self.isolated();
655        let Some((inner_names, inner_lf, inner_schema)) =
656            ctx.resolve_subquery_from(&select.from[0])?
657        else {
658            return Ok(None);
659        };
660        let Some(value) = ctx.try_parse_inner_only_expr(proj, &inner_names, &inner_schema)? else {
661            return Ok(None);
662        };
663        let value = value.meta().undo_aliases();
664
665        let mut corr_preds = Vec::new();
666        let mut local_filters = Vec::new();
667        for conj in MintermIter::new(selection) {
668            if let Some(pred) =
669                scalar_correlation_predicate(conj, &inner_names, &inner_schema, outer_schema)
670            {
671                corr_preds.push(pred);
672            } else if let Some(filter) =
673                ctx.try_parse_inner_only_expr(conj, &inner_names, &inner_schema)?
674            {
675                local_filters.push(filter);
676            } else {
677                return Ok(None);
678            }
679        }
680        // No correlation: the generic `IN` path already handles this.
681        if corr_preds.is_empty() {
682            return Ok(None);
683        }
684
685        let prefix = format_pl_smallstr!("{CORRELATED_COL_PREFIX}{}_", unique_column_name());
686        let set_name = format_pl_smallstr!("{prefix}set");
687
688        let inner_filtered = local_filters.into_iter().fold(inner_lf, LazyFrame::filter);
689        let rename_from: Vec<PlSmallStr> = inner_schema.iter_names().cloned().collect();
690        let rename_to: Vec<PlSmallStr> = rename_from
691            .iter()
692            .map(|name| prefixed_inner(&prefix, name))
693            .collect();
694        let inner_renamed = inner_filtered.rename(&rename_from, &rename_to, true);
695        inner_renamed.set_cached_arena(ctx.lp_arena, ctx.expr_arena);
696
697        let value = value.map_expr(|e| match e {
698            Expr::Column(name) if inner_schema.contains(name.as_str()) => {
699                col(prefixed_inner(&prefix, &name))
700            },
701            other => other,
702        });
703
704        // Grouping collects the candidate values of each outer row into a list.
705        let joined = if corr_preds.iter().all(|p| p.op == SQLBinaryOperator::Eq) {
706            let outer_on: Vec<Expr> = corr_preds.iter().map(|p| col(p.outer.clone())).collect();
707            let inner_on: Vec<Expr> = corr_preds
708                .iter()
709                .map(|p| col(prefixed_inner(&prefix, &p.inner)))
710                .collect();
711            let grouped = inner_renamed
712                .group_by(inner_on.clone())
713                .agg([value.alias(set_name.clone())]);
714            left_join_aggregate(lf, grouped, outer_on, inner_on, &set_name, false)?
715        } else {
716            let join_preds: Vec<Expr> = corr_preds.iter().map(|p| p.to_expr(&prefix)).collect();
717            let idx_name = format_pl_smallstr!("{prefix}idx");
718            let outer_indexed = row_indexed_once(lf, idx_name.clone());
719            let matched = outer_indexed
720                .clone()
721                .join_builder()
722                .with(inner_renamed)
723                .how(JoinType::Inner)
724                .join_where(join_preds);
725            let grouped = matched
726                .group_by([col(idx_name.clone())])
727                .agg([value.alias(set_name.clone())]);
728            left_join_aggregate(
729                outer_indexed,
730                grouped,
731                vec![col(idx_name.clone())],
732                vec![col(idx_name.clone())],
733                &set_name,
734                false,
735            )?
736            .drop(Selector::ByName {
737                names: Arc::from([idx_name]),
738                strict: true,
739            })
740        };
741
742        // An outer row with no matching inner rows joins to null, which is the
743        // empty candidate set rather than an unknown one.
744        let set = col(set_name.clone());
745        let is_in = sql_in_membership(
746            needle.is_in(set.clone(), false),
747            set.clone(),
748            set.clone().is_null().or(set.list().len().eq(lit(0u32))),
749        );
750        // The boolean replaces the candidate list in place.
751        let joined = joined.with_columns([is_in.alias(set_name.clone())]);
752        Ok(Some((joined, set_name)))
753    }
754
755    // Attempt the decorrelation of a single `EXISTS` subquery into a boolean flag
756    // column holding `count(*) > 0` over the correlated match set. An uncorrelated
757    // subquery yields a single constant boolean broadcast onto every outer row
758    // instead, no join being needed to know whether a fixed relation is non-empty.
759    // Returns `None` for shapes that can't be soundly classified.
760    fn try_decorrelate_exists_subquery(
761        &mut self,
762        lf: LazyFrame,
763        outer_schema: &Schema,
764        subquery: &Query,
765    ) -> PolarsResult<Option<(LazyFrame, PlSmallStr)>> {
766        let Some(select) = eligible_subquery_select(subquery) else {
767            return Ok(None);
768        };
769
770        let mut ctx = self.isolated();
771        let Some((inner_names, inner_lf, inner_schema)) =
772            ctx.resolve_subquery_from(&select.from[0])?
773        else {
774            return Ok(None);
775        };
776
777        let mut corr_preds = Vec::new();
778        let mut local_filters = Vec::new();
779        if let Some(selection) = &select.selection {
780            for conj in MintermIter::new(selection) {
781                if let Some(pred) =
782                    scalar_correlation_predicate(conj, &inner_names, &inner_schema, outer_schema)
783                {
784                    corr_preds.push(pred);
785                } else if let Some(filter) =
786                    ctx.try_parse_inner_only_expr(conj, &inner_names, &inner_schema)?
787                {
788                    local_filters.push(filter);
789                } else {
790                    return Ok(None);
791                }
792            }
793        }
794
795        let prefix = format_pl_smallstr!("{CORRELATED_COL_PREFIX}{}_", unique_column_name());
796        let flag_name = format_pl_smallstr!("{prefix}exists");
797        let inner_filtered = local_filters.into_iter().fold(inner_lf, LazyFrame::filter);
798
799        if corr_preds.is_empty() {
800            // Uncorrelated: the inner relation doesn't depend on the outer row, so
801            // its existence is a single constant boolean broadcast onto every row.
802            let inner_flag = inner_filtered.select([len().gt(lit(0)).alias(flag_name.clone())]);
803            inner_flag.set_cached_arena(ctx.lp_arena, ctx.expr_arena);
804            let joined = concat_lf_horizontal(
805                [lf, inner_flag],
806                HConcatOptions {
807                    broadcast_unit_length: true,
808                    ..Default::default()
809                },
810            )?;
811            return Ok(Some((joined, flag_name)));
812        }
813
814        let idx_name = format_pl_smallstr!("{prefix}idx");
815        let count_name = format_pl_smallstr!("{prefix}cnt");
816
817        // Rename inner columns to collision-free names so they can't clash
818        // with outer columns of the same name (as in a self-correlated
819        // `t1 AS x` subquery over outer `t1`).
820        let rename_from: Vec<PlSmallStr> = inner_schema.iter_names().cloned().collect();
821        let rename_to: Vec<PlSmallStr> = rename_from
822            .iter()
823            .map(|name| prefixed_inner(&prefix, name))
824            .collect();
825        let inner_renamed = inner_filtered.rename(&rename_from, &rename_to, true);
826        inner_renamed.set_cached_arena(ctx.lp_arena, ctx.expr_arena);
827
828        let join_preds: Vec<Expr> = corr_preds.iter().map(|p| p.to_expr(&prefix)).collect();
829
830        let outer_indexed = row_indexed_once(lf, idx_name.clone());
831        let matched = outer_indexed
832            .clone()
833            .join_builder()
834            .with(inner_renamed)
835            .how(JoinType::Inner)
836            .join_where(join_preds);
837        let grouped = matched
838            .group_by([col(idx_name.clone())])
839            .agg([len().alias(count_name.clone())]);
840
841        let joined = outer_indexed
842            .join_builder()
843            .with(grouped)
844            .left_on([col(idx_name.clone())])
845            .right_on([col(idx_name.clone())])
846            .how(JoinType::Left)
847            .coalesce(JoinCoalesce::CoalesceColumns)
848            .maintain_order(MaintainOrderJoin::Left)
849            .finish()?
850            .with_columns([col(count_name.clone())
851                .fill_null(lit(0))
852                .gt(lit(0))
853                .alias(flag_name.clone())])
854            .drop(Selector::ByName {
855                names: Arc::from([idx_name, count_name]),
856                strict: true,
857            });
858        Ok(Some((joined, flag_name)))
859    }
860
861    #[cfg(not(feature = "semi_anti_join"))]
862    fn try_rewrite_exists_as_join(
863        &mut self,
864        _lf: &LazyFrame,
865        _subquery: &Query,
866        _negated: bool,
867        _outer_schema: &Schema,
868    ) -> PolarsResult<Option<LazyFrame>> {
869        Ok(None)
870    }
871
872    #[cfg(not(feature = "semi_anti_join"))]
873    fn try_rewrite_in_subquery_as_join(
874        &mut self,
875        _lf: &LazyFrame,
876        _lhs: &SQLExpr,
877        _subquery: &Query,
878        _anti: bool,
879        _filter_mode: FilterMode,
880        _outer_schema: &Schema,
881    ) -> PolarsResult<Option<LazyFrame>> {
882        Ok(None)
883    }
884}

886// Semi/anti join the outer frame against the (filtered, arena-cached) inner.
887#[cfg(feature = "semi_anti_join")]
888fn build_semi_anti_join(
889    lf: &LazyFrame,
890    inner_lf: LazyFrame,
891    left_on: Vec<Expr>,
892    right_on: Vec<Expr>,
893    anti: bool,
894) -> PolarsResult<LazyFrame> {
895    let join_type = if anti { JoinType::Anti } else { JoinType::Semi };
896    lf.clone()
897        .join_builder()
898        .with(inner_lf)
899        .left_on(left_on)
900        .right_on(right_on)
901        .how(join_type)
902        .finish()
903}

905// Account for 3VL interaction with NULL values
906#[cfg(feature = "semi_anti_join")]
907fn refine_not_in_anti_join(
908    joined: LazyFrame,
909    inner_lf: LazyFrame,
910    left_key: &Expr,
911    right_key: &Expr,
912    corr_outer: &[Expr],
913    corr_inner: &[Expr],
914) -> PolarsResult<LazyFrame> {
915    if corr_inner.is_empty() {
916        // Uncorrelated
917        let flag_name = unique_column_name();
918        let flag = when(len().eq(lit(0u32)))
919            .then(lit(NULL).cast(DataType::Boolean))
920            .otherwise(right_key.clone().is_null().any(true))
921            .alias(flag_name.clone());
922        let keep = when(col(flag_name.clone()).is_null())
923            .then(lit(true)) // empty set
924            .when(col(flag_name.clone())) // set has a NULL
925            .then(lit(false))
926            .otherwise(left_key.clone().is_not_null());
927
928        return Ok(joined
929            .join_builder()
930            .with(inner_lf.select([flag]))
931            .how(JoinType::Cross)
932            .finish()?
933            .filter(keep)
934            .drop(Selector::ByName {
935                names: [flag_name].into(),
936                strict: true,
937            }));
938    }
939
940    // Correlated
941    let corr_keys = |lf: LazyFrame| lf.select(corr_inner).unique(None, UniqueKeepStrategy::Any);
942    let exclude_groups = |rows: LazyFrame, groups: LazyFrame| {
943        rows.join_builder()
944            .with(groups)
945            .left_on(corr_outer)
946            .right_on(corr_inner)
947            .how(JoinType::Anti)
948            .finish()
949    };
950    let kept_non_null = exclude_groups(
951        joined.clone().filter(left_key.clone().is_not_null()),
952        corr_keys(inner_lf.clone().filter(right_key.clone().is_null())),
953    )?;
954    let kept_null = exclude_groups(
955        joined.filter(left_key.clone().is_null()),
956        corr_keys(inner_lf),
957    )?;
958
959    Ok(concat(
960        [kept_non_null, kept_null],
961        UnionArgs {
962            rechunk: false,
963            parallel: true,
964            ..Default::default()
965        },
966    )
967    .expect("'NOT IN' 3VL union has identical schemas"))
968}

970/// An iterator over all the minterms in an SQL boolean expression: the terms
971/// that `AND` together to form it, descending through parenthesized `Nested`
972/// expressions. The SQL-AST analogue of the `AExpr`-level
973/// `polars_plan::plans::aexpr::MintermIter`.
974struct MintermIter<'a> {
975    stack: Vec<&'a SQLExpr>,
976}
977
978impl<'a> Iterator for MintermIter<'a> {
979    type Item = &'a SQLExpr;
980
981    fn next(&mut self) -> Option<Self::Item> {
982        let mut top = self.stack.pop()?;
983        loop {
984            match top {
985                SQLExpr::Nested(inner) => top = inner,
986                SQLExpr::BinaryOp {
987                    left,
988                    op: SQLBinaryOperator::And,
989                    right,
990                } => {
991                    self.stack.push(right);
992                    top = left;
993                },
994                _ => return Some(top),
995            }
996        }
997    }
998}
999
1000impl<'a> MintermIter<'a> {
1001    fn new(root: &'a SQLExpr) -> Self {
1002        Self { stack: vec![root] }
1003    }
1004}

1021// An equi-correlation conjunct `inner.col = outer.col` (either way round) as a
1022// `(outer key, inner key)` column-name pair, or `None` when the conjunct is
1023// anything else (non-equality, unresolvable names, both columns on the same
1024// side).
1025#[cfg(feature = "semi_anti_join")]
1026fn correlation_key_pair(
1027    conj: &SQLExpr,
1028    inner_names: &PlHashSet<String>,
1029    inner_schema: &Schema,
1030    outer_schema: &Schema,
1031) -> Option<(PlSmallStr, PlSmallStr)> {
1032    let SQLExpr::BinaryOp {
1033        left,
1034        op: SQLBinaryOperator::Eq,
1035        right,
1036    } = conj
1037    else {
1038        return None;
1039    };
1040    let (lside, lname) =
1041        classify_correlation_column(left, inner_names, inner_schema, outer_schema)?;
1042    let (rside, rname) =
1043        classify_correlation_column(right, inner_names, inner_schema, outer_schema)?;
1044    match (lside, rside) {
1045        (CorrelationSide::Outer, CorrelationSide::Inner) => Some((lname, rname)),
1046        (CorrelationSide::Inner, CorrelationSide::Outer) => Some((rname, lname)),
1047        _ => None,
1048    }
1049}

1051// Whether every column the expression references resolves to the inner relation,
1052// so it can be evaluated against the inner frame alone. A qualified name resolves
1053// through its qualifier: only an inner relation's name or alias counts as inner.
1054// An unqualified name binds to the innermost scope that has it, so the inner
1055// schema wins over the outer one.
1056fn binds_to_inner_relation(
1057    expr: &SQLExpr,
1058    inner_names: &PlHashSet<String>,
1059    inner_schema: &Schema,
1060) -> bool {
1061    visit_expressions(expr, |e| {
1062        let resolves = match e {
1063            // A nested subquery is its own scope, which this cannot resolve against.
1064            _ if is_subquery_expr(e) => false,
1065            SQLExpr::Identifier(_) | SQLExpr::CompoundIdentifier(_) => qualifier_and_name(e)
1066                .is_some_and(|(qualifier, name)| {
1067                    qualifier.is_none_or(|q| inner_names.contains(q)) && inner_schema.contains(name)
1068                }),
1069            _ => true,
1070        };
1071        if resolves {
1072            ControlFlow::Continue(())
1073        } else {
1074            ControlFlow::Break(())
1075        }
1076    })
1077    .is_continue()
1078}

1080// Split an identifier into its optional table qualifier and its bare column name.
1081// Catalog and schema prefixes are dropped, matching how `get_table_name` builds
1082// `inner_names`.
1083fn qualifier_and_name(expr: &SQLExpr) -> Option<(Option<&str>, &str)> {
1084    match expr {
1085        SQLExpr::Identifier(ident) => Some((None, ident.value.as_str())),
1086        SQLExpr::CompoundIdentifier(parts) => {
1087            let (last, init) = parts.split_last()?;
1088            Some((init.last().map(|q| q.value.as_str()), last.value.as_str()))
1089        },
1090        _ => None,
1091    }
1092}

1094// Classify a correlation operand as an inner- or outer-query column and return
1095// its bare name. A qualified identifier (`tbl.col`) resolves by its qualifier:
1096// an inner relation's name/alias means inner, anything else means outer (so
1097// same-named columns like `o.id = c.id` resolve). An unqualified identifier
1098// resolves by schema membership. `None` for non-identifiers or names that can't
1099// be placed (in neither schema, or ambiguous).
1100fn classify_correlation_column(
1101    expr: &SQLExpr,
1102    inner_names: &PlHashSet<String>,
1103    inner_schema: &Schema,
1104    outer_schema: &Schema,
1105) -> Option<(CorrelationSide, PlSmallStr)> {
1106    let (qualifier, name) = qualifier_and_name(expr)?;
1107    let name: PlSmallStr = name.into();
1108    match qualifier {
1109        Some(q) if inner_names.contains(q) => inner_schema
1110            .contains(name.as_str())
1111            .then_some((CorrelationSide::Inner, name)),
1112        // Any other qualifier is taken as outer: the outer query may span
1113        // several relations and their names/aliases aren't visible here, so
1114        // only the column's schema membership can be checked.
1115        Some(_) => outer_schema
1116            .contains(name.as_str())
1117            .then_some((CorrelationSide::Outer, name)),
1118        None => match (
1119            inner_schema.contains(name.as_str()),
1120            outer_schema.contains(name.as_str()),
1121        ) {
1122            (true, false) => Some((CorrelationSide::Inner, name)),
1123            (false, true) => Some((CorrelationSide::Outer, name)),
1124            _ => None,
1125        },
1126    }
1127}

1129// Shared eligibility gate for the rewrites: bail on any clause that changes
1130// which rows the subquery yields. Exhaustive destructuring (no `..`) is on
1131// purpose: a new sqlparser clause must not compile until it gets an explicit
1132// keep-or-bail decision here.
1133fn eligible_subquery_select(subquery: &Query) -> Option<&Select> {
1134    let Query {
1135        with, // CTEs aren't resolved inside the rewrite: bail
1136        body,
1137        order_by: _,      // row order can't affect existence/membership
1138        limit_clause,     // LIMIT/OFFSET change the yielded rows: bail
1139        fetch,            // FETCH FIRST is LIMIT spelled differently: bail
1140        locks: _,         // row locking doesn't change the rows
1141        for_clause,       // FOR XML/JSON reshape the result: bail
1142        settings,         // ClickHouse SETTINGS can change results: bail
1143        format_clause: _, // output serialization only
1144        pipe_operators,   // `|>` operators transform the rows: bail
1145    } = subquery;
1146    if with.is_some()
1147        || limit_clause.is_some()
1148        || fetch.is_some()
1149        || for_clause.is_some()
1150        || settings.is_some()
1151        || !pipe_operators.is_empty()
1152    {
1153        return None;
1154    }
1155    let SetExpr::Select(select) = body.as_ref() else {
1156        return None;
1157    };
1158    let Select {
1159        select_token: _,
1160        // Deduplication is existence/membership-invariant; the IN rewrite
1161        // separately bails on `DISTINCT ON`, which is not.
1162        distinct: _,
1163        top,                    // TOP is LIMIT spelled differently: bail
1164        top_before_distinct: _, // only meaningful with `top`
1165        // The projection is validated by the callers: EXISTS ignores it, IN
1166        // requires a single plain expression (which also rules out wildcards
1167        // and the `exclude` modifier).
1168        projection: _,
1169        exclude: _,
1170        into,          // SELECT INTO is not a pure subquery: bail
1171        from,          // must be one (possibly joined) relation
1172        lateral_views, // row-multiplying: bail
1173        prewhere,      // an extra filter we don't fold in: bail
1174        selection: _,  // split into join keys/filters by callers
1175        group_by,      // aggregation changes the yielded rows: bail
1176        cluster_by: _, // layout/order hints: row-set preserving
1177        distribute_by: _,
1178        sort_by: _,
1179        having,                   // aggregation filter: bail
1180        named_window: _,          // definitions only; uses are parsed later
1181        qualify,                  // post-window filter changes the rows: bail
1182        window_before_qualify: _, // only meaningful with `qualify`
1183        value_table_mode,         // changes what a row is: bail
1184        connect_by,               // hierarchical recursion: bail
1185        optimizer_hints,          // unsupported: bail
1186        select_modifiers,         // unsupported: bail
1187        flavor: _,                // surface syntax only
1188    } = select.as_ref();
1189    let no_group_by = matches!(
1190        group_by,
1191        GroupByExpr::Expressions(e, m) if e.is_empty() && m.is_empty()
1192    );
1193    if from.len() != 1
1194        || !no_group_by
1195        || top.is_some()
1196        || into.is_some()
1197        || having.is_some()
1198        || qualify.is_some()
1199        || prewhere.is_some()
1200        || !connect_by.is_empty()
1201        || value_table_mode.is_some()
1202        || !lateral_views.is_empty()
1203        || !optimizer_hints.is_empty()
1204        || select_modifiers.is_some()
1205    {
1206        return None;
1207    }
1208    Some(select)
1209}

1211/// Correlated subqueries already lowered against a frame, each mapped to the column its
1212/// decorrelation materialised. The `bool` is the `exists` flag: a scalar subquery and an
1213/// `EXISTS` over the same inner query materialise different columns and must not share a
1214/// binding.
1215///
1216/// A set is only valid while its columns are still present and still mean the same thing
1217/// per row, so it may be shared across passes over one frame but must be started afresh
1218/// once the frame has been re-projected or aggregated.
1219pub(crate) type SubqueryBindings = Vec<(Query, SubqueryKind, PlSmallStr)>;
1220
1221/// Which predicate a lowered subquery column answers.
1222#[derive(Clone, PartialEq)]
1223pub(crate) enum SubqueryKind {
1224    Scalar,
1225    Exists,
1226    /// `IN`, keyed by its left-hand side: one subquery can serve several.
1227    In(Box<SQLExpr>),
1228}

1230/// Whether a subquery reads a qualifier it does not declare, which is what
1231/// correlates it with the query around it.
1232pub(crate) fn is_correlated_subquery(query: &Query) -> bool {
1233    #[derive(Default)]
1234    struct Declared(PlHashSet<String>);
1235
1236    impl Visitor for Declared {
1237        type Break = ();
1238
1239        fn pre_visit_table_factor(&mut self, factor: &TableFactor) -> ControlFlow<()> {
1240            if let Some(name) = get_table_name(factor) {
1241                self.0.insert(name);
1242            }
1243            ControlFlow::Continue(())
1244        }
1245    }
1246
1247    let mut declared = Declared::default();
1248    let _ = query.visit(&mut declared);
1249
1250    struct Foreign<'a>(&'a PlHashSet<String>);
1251
1252    impl Visitor for Foreign<'_> {
1253        type Break = ();
1254
1255        fn pre_visit_expr(&mut self, expr: &SQLExpr) -> ControlFlow<()> {
1256            match qualifier_and_name(expr) {
1257                Some((Some(qualifier), _)) if !self.0.contains(qualifier) => ControlFlow::Break(()),
1258                _ => ControlFlow::Continue(()),
1259            }
1260        }
1261    }
1262
1263    query.visit(&mut Foreign(&declared.0)).is_break()
1264}

1266/// Rewrite `x = ANY (subquery)` to `x IN (subquery)` and `x <> ALL (subquery)`
1267/// to `x NOT IN (subquery)`, which they are equivalent to.
1268struct DesugarQuantified;
1269
1270impl VisitorMut for DesugarQuantified {
1271    type Break = ();
1272
1273    fn pre_visit_expr(&mut self, expr: &mut SQLExpr) -> ControlFlow<()> {
1274        let (left, right, negated) = match &*expr {
1275            SQLExpr::AnyOp {
1276                left,
1277                compare_op: SQLBinaryOperator::Eq,
1278                right,
1279                ..
1280            } => (left, right, false),
1281            SQLExpr::AllOp {
1282                left,
1283                compare_op: SQLBinaryOperator::NotEq,
1284                right,
1285            } => (left, right, true),
1286            _ => return ControlFlow::Continue(()),
1287        };
1288        let SQLExpr::Subquery(subquery) = right.as_ref() else {
1289            return ControlFlow::Continue(());
1290        };
1291        *expr = SQLExpr::InSubquery {
1292            expr: left.clone(),
1293            subquery: subquery.clone(),
1294            negated,
1295        };
1296        ControlFlow::Continue(())
1297    }
1298}
1299
1300pub(crate) fn desugar_quantified_subqueries(stmt: &mut Statement) {
1301    let _ = VisitMut::visit(stmt, &mut DesugarQuantified);
1302}

1304/// Which correlated-subquery node kinds a lowering pass should claim.
1305#[derive(Clone, Copy, PartialEq, Eq)]
1306pub(crate) enum LowerScope {
1307    /// Scalar subqueries only, leaving predicate subqueries for the semi/anti join rewrite.
1308    ScalarOnly,
1309    /// Scalar subqueries and the predicate subqueries `[NOT] EXISTS` and `[NOT] IN`.
1310    ScalarAndPredicates,
1311}
1312
1313fn lowerable_kind(expr: &SQLExpr, scope: LowerScope) -> bool {
1314    match expr {
1315        SQLExpr::Subquery(_) => true,
1316        SQLExpr::Exists { .. } | SQLExpr::InSubquery { .. } => {
1317            scope == LowerScope::ScalarAndPredicates
1318        },
1319        _ => false,
1320    }
1321}

1323/// Rewrites correlated subqueries in-place into references to the columns their
1324/// decorrelated lowering materialises, threading the growing `LazyFrame` through the
1325/// traversal. `query_depth` keeps each subquery a leaf rather than recursing into it.
1326struct CorrelatedLowering<'a> {
1327    ctx: &'a mut SQLContext,
1328    lf: LazyFrame,
1329    outer_schema: &'a Schema,
1330    scope: LowerScope,
1331    bindings: &'a mut SubqueryBindings,
1332    query_depth: usize,
1333    changed: bool,
1334}
1335
1336impl CorrelatedLowering<'_> {
1337    /// Lower `subquery` (or reuse an earlier identical lowering), returning the
1338    /// materialised column name, or `None` if it can't be soundly lowered.
1339    fn lower(&mut self, subquery: &Query, kind: SubqueryKind) -> PolarsResult<Option<PlSmallStr>> {
1340        if let Some((_, _, name)) = self
1341            .bindings
1342            .iter()
1343            .find(|(q, bound, _)| *bound == kind && q == subquery)
1344        {
1345            return Ok(Some(name.clone()));
1346        }
1347        // The frame is cloned rather than moved so it survives a declined lowering.
1348        let lowered = match &kind {
1349            SubqueryKind::Exists => self.ctx.try_decorrelate_exists_subquery(
1350                self.lf.clone(),
1351                self.outer_schema,
1352                subquery,
1353            )?,
1354            SubqueryKind::Scalar => self.ctx.try_decorrelate_scalar_subquery(
1355                self.lf.clone(),
1356                self.outer_schema,
1357                subquery,
1358            )?,
1359            SubqueryKind::In(lhs) => self.ctx.try_decorrelate_in_subquery(
1360                self.lf.clone(),
1361                self.outer_schema,
1362                lhs,
1363                subquery,
1364            )?,
1365        };
1366        let Some((new_lf, name)) = lowered else {
1367            return Ok(None);
1368        };
1369        self.lf = new_lf;
1370        self.bindings.push((subquery.clone(), kind, name.clone()));
1371        Ok(Some(name))
1372    }
1373}
1374
1375impl VisitorMut for CorrelatedLowering<'_> {
1376    type Break = PolarsError;
1377
1378    fn pre_visit_query(&mut self, _query: &mut Query) -> ControlFlow<PolarsError> {
1379        self.query_depth += 1;
1380        ControlFlow::Continue(())
1381    }
1382
1383    fn post_visit_query(&mut self, _query: &mut Query) -> ControlFlow<PolarsError> {
1384        self.query_depth -= 1;
1385        ControlFlow::Continue(())
1386    }
1387
1388    fn pre_visit_expr(&mut self, expr: &mut SQLExpr) -> ControlFlow<PolarsError> {
1389        // Inside a nested query: not ours to lower against the outer schema.
1390        if self.query_depth > 0 || !lowerable_kind(expr, self.scope) {
1391            return ControlFlow::Continue(());
1392        }
1393        let (subquery, kind, negated) = match &*expr {
1394            SQLExpr::Subquery(subquery) => (subquery.as_ref(), SubqueryKind::Scalar, false),
1395            SQLExpr::Exists { subquery, negated } => {
1396                (subquery.as_ref(), SubqueryKind::Exists, *negated)
1397            },
1398            SQLExpr::InSubquery {
1399                expr: lhs,
1400                subquery,
1401                negated,
1402            } => (subquery.as_ref(), SubqueryKind::In(lhs.clone()), *negated),
1403            _ => unreachable!("guarded by lowerable_kind"),
1404        };
1405        let name = match self.lower(subquery, kind) {
1406            Ok(Some(name)) => name,
1407            Ok(None) => return ControlFlow::Continue(()),
1408            Err(e) => return ControlFlow::Break(e),
1409        };
1410        // `EXISTS` is never NULL; `IN` may be, and negating a NULL keeps it NULL,
1411        // which is what SQL asks for.
1412        let resolved = SQLExpr::Identifier(Ident::new(name.as_str()));
1413        *expr = if negated {
1414            SQLExpr::UnaryOp {
1415                op: SQLUnaryOperator::Not,
1416                expr: Box::new(resolved),
1417            }
1418        } else {
1419            resolved
1420        };
1421        self.changed = true;
1422        ControlFlow::Continue(())
1423    }
1424}

1426// Whether an expression can serve as a join key. Join keys must be elementwise,
1427// so that every key is as long as the frame it is built from. Anything this does
1428// not recognise declines the rewrite, which costs an optimisation rather than
1429// correctness.
1430fn usable_as_join_key(expr: &Expr) -> bool {
1431    match expr {
1432        Expr::Column(_) | Expr::Literal(_) => true,
1433        Expr::Alias(inner, _) | Expr::Cast { expr: inner, .. } => usable_as_join_key(inner),
1434        Expr::BinaryExpr { left, op: _, right } => {
1435            usable_as_join_key(left) && usable_as_join_key(right)
1436        },
1437        Expr::Ternary {
1438            predicate,
1439            truthy,
1440            falsy,
1441        } => {
1442            usable_as_join_key(predicate) && usable_as_join_key(truthy) && usable_as_join_key(falsy)
1443        },
1444        _ => false,
1445    }
1446}

1459// Left-join a per-group aggregate onto the outer frame, filling unmatched `COUNT`
1460// results with 0 (every other aggregate stays NULL for unmatched rows).
1461fn left_join_aggregate(
1462    outer: LazyFrame,
1463    grouped: LazyFrame,
1464    left_on: Vec<Expr>,
1465    right_on: Vec<Expr>,
1466    result_name: &PlSmallStr,
1467    count_like: bool,
1468) -> PolarsResult<LazyFrame> {
1469    let joined = outer
1470        .join_builder()
1471        .with(grouped)
1472        .left_on(left_on)
1473        .right_on(right_on)
1474        .how(JoinType::Left)
1475        .coalesce(JoinCoalesce::CoalesceColumns)
1476        .maintain_order(MaintainOrderJoin::Left)
1477        .finish()?;
1478    Ok(if count_like {
1479        joined.with_columns([col(result_name.clone()).fill_null(lit(0))])
1480    } else {
1481        joined
1482    })
1483}

1485// Peel the alias/cast wrappers SQL puts around an aggregate to reach the
1486// aggregate that determines the empty-group value.
1487fn agg_output_root(expr: &Expr) -> &Expr {
1488    match expr {
1489        Expr::Alias(inner, _) => agg_output_root(inner.as_ref()),
1490        Expr::Cast { expr: inner, .. } => agg_output_root(inner.as_ref()),
1491        other => other,
1492    }
1493}

1495// A correlation conjunct `<inner col> <cmp> <outer col>` (either way round),
1496// tracking which side is the inner column so the comparison can be rebuilt in
1497// its original orientation as a join predicate.
1498struct CorrPredicate {
1499    outer: PlSmallStr,
1500    inner: PlSmallStr,
1501    op: SQLBinaryOperator,
1502    inner_on_left: bool,
1503}
1504
1505impl CorrPredicate {
1506    fn to_expr(&self, prefix: &str) -> Expr {
1507        let outer = col(self.outer.clone());
1508        let inner = col(prefixed_inner(prefix, &self.inner));
1509        let (l, r) = if self.inner_on_left {
1510            (inner, outer)
1511        } else {
1512            (outer, inner)
1513        };
1514        match self.op {
1515            SQLBinaryOperator::Eq => l.eq(r),
1516            SQLBinaryOperator::NotEq => l.neq(r),
1517            SQLBinaryOperator::Lt => l.lt(r),
1518            SQLBinaryOperator::LtEq => l.lt_eq(r),
1519            SQLBinaryOperator::Gt => l.gt(r),
1520            SQLBinaryOperator::GtEq => l.gt_eq(r),
1521            // Guarded by `scalar_correlation_predicate`.
1522            _ => unreachable!("non-comparison correlation operator"),
1523        }
1524    }
1525}
1526
1527// A comparison conjunct with one bare inner column and one bare outer column,
1528// or `None` for anything else. Both sides must be simple columns; more complex
1529// correlation shapes leave the whole subquery to the generic path.
1530fn scalar_correlation_predicate(
1531    conj: &SQLExpr,
1532    inner_names: &PlHashSet<String>,
1533    inner_schema: &Schema,
1534    outer_schema: &Schema,
1535) -> Option<CorrPredicate> {
1536    let SQLExpr::BinaryOp { left, op, right } = conj else {
1537        return None;
1538    };
1539    if !matches!(
1540        op,
1541        SQLBinaryOperator::Eq
1542            | SQLBinaryOperator::NotEq
1543            | SQLBinaryOperator::Lt
1544            | SQLBinaryOperator::LtEq
1545            | SQLBinaryOperator::Gt
1546            | SQLBinaryOperator::GtEq
1547    ) {
1548        return None;
1549    }
1550    let (lside, lname) =
1551        classify_correlation_column(left, inner_names, inner_schema, outer_schema)?;
1552    let (rside, rname) =
1553        classify_correlation_column(right, inner_names, inner_schema, outer_schema)?;
1554    match (lside, rside) {
1555        (CorrelationSide::Inner, CorrelationSide::Outer) => Some(CorrPredicate {
1556            outer: rname,
1557            inner: lname,
1558            op: op.clone(),
1559            inner_on_left: true,
1560        }),
1561        (CorrelationSide::Outer, CorrelationSide::Inner) => Some(CorrPredicate {
1562            outer: lname,
1563            inner: rname,
1564            op: op.clone(),
1565            inner_on_left: false,
1566        }),
1567        _ => None,
1568    }
